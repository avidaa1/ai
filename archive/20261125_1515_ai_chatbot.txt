function archGetKey(){return localStorage.getItem('avidaa_anthropic_key')||'';}

function archActivate(){
  var k=document.getElementById('arch-key-input').value.trim();
  if(!k){archKeyError('Enter a valid Anthropic API key.');return;}
  if(!k.startsWith('sk-ant')){archKeyError('Key should start with sk-ant...');return;}
  localStorage.setItem('avidaa_anthropic_key',k);
  document.getElementById('arch-key-overlay').style.display='none';
}

function archKeyError(msg){
  var inp=document.getElementById('arch-key-input');
  inp.style.borderColor='var(--volatile)';
  inp.placeholder=msg;
  setTimeout(function(){inp.style.borderColor='var(--amber-b)';inp.placeholder='sk-ant-api03-...';},2200);
}

function archTimestamp(){
  var d=new Date();
  return ('0'+d.getHours()).slice(-2)+':'+('0'+d.getMinutes()).slice(-2);
}

function archRender(text){
  var t=text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  t=t.replace(/^### (.+)$/gm,'<div style="font-family:var(--cond);font-weight:700;font-size:15px;letter-spacing:1px;color:var(--amber);margin:10px 0 4px">$1</div>');
  t=t.replace(/^## (.+)$/gm,'<div style="font-family:var(--cond);font-weight:700;font-size:17px;letter-spacing:1px;color:var(--cyan);margin:12px 0 4px">$1</div>');
  t=t.replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>');
  t=t.replace(/`([^`]+)`/g,'<code>$1</code>');
  t=t.replace(/^\d+\. (.+)$/gm,'<div style="padding:2px 0 2px 14px;border-left:2px solid var(--amber-b);margin:2px 0"><span style="color:var(--amber)">&#9656;</span> $1</div>');
  t=t.replace(/^[-*] (.+)$/gm,'<div style="padding:2px 0 2px 14px;border-left:2px solid var(--cyan-b);margin:2px 0"><span style="color:var(--cyan)">&#8729;</span> $1</div>');
  t=t.replace(/\n/g,'<br>');
  return t;
}

async function archSend(){
  var inp=document.getElementById('arch-input');
  var text=inp.value.trim();
  if(!text||archBusy)return;
  var key=archGetKey();
  if(!key){archChangeKey();return;}
  archBusy=true;
  inp.value='';inp.style.height='auto';
  document.getElementById('arch-send-btn').disabled=true;
  archAppendMsg('user',text);
  archHist.push({role:'user',content:text});
  archShowTyping();
  try{
    var res=await fetch('https://api.anthropic.com/v1/messages',{
      method:'POST',
      headers:{
        'Content-Type':'application/json',
        'x-api-key':key,
        'anthropic-version':'2023-06-01',
        'anthropic-dangerous-direct-browser-access':'true'
      },
      body:JSON.stringify({
        model:'claude-sonnet-4-6',
        max_tokens:1500,
        system:buildArchSystem(),
        messages:archHist
      })
    });
    var data=await res.json();
    archHideTyping();
    if(data.content&&data.content[0]&&data.content[0].text){
      var reply=data.content[0].text;
      archHist.push({role:'assistant',content:reply});
      if(archHist.length>30)archHist=archHist.slice(-30); // keep last 24 messages
      archAppendMsg('assistant',reply);
    } else if(data.error){
      var em=data.error.message||'Unknown error';
      archAppendMsg('assistant','[API ERROR] '+em+'\n\nCheck your API key or rate limits.');
      if(data.error.type==='authentication_error')setTimeout(archChangeKey,1200);
    } else {
      archAppendMsg('assistant','[ARCH] No response received. Check API key and try again.');
    }
  } catch(e){
    archHideTyping();
    archAppendMsg('assistant','[NETWORK ERROR] '+e.message+'\n\nCheck your connection and API key.');
  }
  archBusy=false;
  document.getElementById('arch-send-btn').disabled=false;
  document.getElementById('arch-input').focus();
}

  archSend();
}

    {name:'cheapest',patterns:['cheap','cheapest','low cost','affordable','budget','price','cost']},
    {name:'fastest',patterns:['fast','fastest','speed','quick','rapid','tok/s']},
    {name:'biggest',patterns:['big','biggest','large','largest','most parameter','most powerful']},
    {name:'list',patterns:['list','all','show me','what are','give me']},
    {name:'context',patterns:['context','window','token','length']},

// Render markdown to HTML
function intelRender(text){
  var t=text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  t=t.replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>');
  t=t.replace(/`([^`]+)`/g,'<code>$1</code>');
  t=t.replace(/\n\n/g,'<br><br>');
  t=t.replace(/\n- /g,'<br>• ');
  t=t.replace(/\n\|/g,'<br>|');
  t=t.replace(/\n/g,'<br>');
  return t;
}
